152 template <bound_struct Tstruct>
153 void _load_struct(const Value& value, Tstruct& target, const string_t& path)
154 {
155 if (!value.is_mapping())
156 {
157 _error(u8"A mapping is expected here.", path);
158 return;
159 }
160
161 const mapping_t& mapping = value.as_mapping();
162
163 std::apply([&](const auto&... fields)
164 {
165 (_load_field(mapping, target, fields, path), ...);
166 }, Binding<Tstruct>::fields);
167
168 _report_unknown_keys<Tstruct>(mapping, path);
169 }

171 template <typename Tstruct, typename Tfield>
172 void _load_field(const mapping_t& mapping, Tstruct& target, const Tfield& field, const string_t& path)
173 {
174 const auto it = mapping.find(field.key);
175
176 // A field the value does not mention keeps what it had.
177 if (it == mapping.end())
178 return;
179
180 _load_value(it->second, target.*(field.member), _extend(path, field.key));
181 }

183 template <typename Tstruct>
184 void _report_unknown_keys(const mapping_t& mapping, const string_t& path)
185 {
186 for (const auto& p : mapping)
187 {
188 bool known = false;
189
190 std::apply([&](const auto&... fields)
191 {
192 ((known = known || fields.key == p.first), ...);
193 }, Binding<Tstruct>::fields);
194
195 if (!known)
196 _warning(u8"Unknown key.", _extend(path, p.first));
197 }

200 template <typename T>
201 void _load_value(const Value& value, T& target, const string_t& path)
202 {
203 if constexpr (detail::bound_optional<T>)
204 _load_optional(value, target, path);
205 else if constexpr (bound_struct<T>)
206 _load_struct(value, target, path);
207 else if constexpr (named_enum<T>)
208 _load_enum(value, target, path);
209 else if constexpr (std::same_as<T, string_t>)
210 _load_string(value, target, path);
211 else if constexpr (detail::bound_sequence<T>)
212 _load_sequence(value, target, path);
213 else if constexpr (detail::bound_map<T>)
214 _load_map(value, target, path);
215 else if constexpr (string_convertible<T>)
216 _load_string_convertible(value, target, path);
217 else if constexpr (std::same_as<T, bool>)
218 _load_boolean(value, target, path);
219 else if constexpr (std::integral<T>)
220 _load_integer(value, target, path);
221 else if constexpr (std::floating_point<T>)
222 _load_float(value, target, path);
223 else
224 static_assert(detail::always_false<T>, "This type cannot be bound: give it a `Binding` or an `EnumNames` specialization.");
225 }

233 template <detail::bound_optional Toptional>
234 void _load_optional(const Value& value, Toptional& target, const string_t& path)
235 {
236 if (value.is_null())
237 {
238 target.reset();
239 return;
240 }
241
242 typename Toptional::value_type inner = target.has_value() ? *target : typename Toptional::value_type{};
243 _load_value(value, inner, path);
244 target = std::move(inner);

294 template <detail::bound_sequence Tsequence>
295 void _load_sequence(const Value& value, Tsequence& target, const string_t& path)
296 {
297 if (!value.is_sequence())
298 {
299 _error(u8"A sequence is expected here.", path);
300 return;
301 }
302
303 // A sequence is replaced rather than added to: a configuration stating a
304 // list means that list, not that list on top of the default one.
305 target.clear();
306
307 const sequence_t& sequence = value.as_sequence();
308
309 for (size_t i = 0; i < sequence.size(); ++i)
310 {
311 typename Tsequence::value_type entry{};
312 _load_value(sequence[i], entry, _index(path, i));
313 target.push_back(std::move(entry));
314 }
315 }

380 template <bound_struct Tstruct>
381 Value _save_struct(const Tstruct& source)
382 {
383 mapping_t mapping;
384
385 std::apply([&](const auto&... fields)
386 {
387 (_save_field(mapping, source, fields), ...);
388 }, Binding<Tstruct>::fields);
389
390 return Value{std::move(mapping)};
391 }

393 template <typename Tstruct, typename Tfield>
394 void _save_field(mapping_t& mapping, const Tstruct& source, const Tfield& field)
395 {
396 const auto& member = source.*(field.member);
397
398 // Left out entirely rather than written as an explicit null:
399 // a field not set at this level is the normal case, and a
400 // hand-authored file should only ever have to state what differs.
401 if constexpr (detail::bound_optional<std::remove_cvref_t<decltype(member)>>)
402 {
403 if (!member.has_value())
404 return;
405 }
406
407 mapping[string_t{field.key}] = _save_value(member);
408 }

410 template <typename T>
411 Value _save_value(const T& source)
412 {
413 if constexpr (detail::bound_optional<T>)
414 return _save_optional(source);
415 else if constexpr (bound_struct<T>)
416 return _save_struct(source);
417 else if constexpr (named_enum<T>)
418 return _save_enum(source);
419 else if constexpr (std::same_as<T, string_t>)
420 return Value{source};
421 else if constexpr (detail::bound_sequence<T>)
422 return _save_sequence(source);
423 else if constexpr (detail::bound_map<T>)
424 return _save_map(source);
425 else if constexpr (string_convertible<T>)
426 return _save_string_convertible(source);
427 else if constexpr (std::same_as<T, bool>)
428 return Value{source};
429 else if constexpr (std::integral<T>)
430 return Value{static_cast<integer_t>(source)};
431 else if constexpr (std::floating_point<T>)
432 return Value{static_cast<float_t>(source)};
433 else
434 static_assert(detail::always_false<T>, "This type cannot be bound: give it a `Binding` or an `EnumNames` specialization.");
435 return {};
436 }

458 template <detail::bound_sequence Tsequence>
459 Value _save_sequence(const Tsequence& source)
460 {
461 sequence_t sequence;
462 sequence.reserve(source.size());
463
464 for (const auto& entry : source)
465 sequence.push_back(_save_value(entry));
466
467 return Value{std::move(sequence)};
468 }

473 template <detail::bound_map Tmap>

477
