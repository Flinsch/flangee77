97 template <bound_struct Tstruct>
98 void _load_struct(const Value& value, Tstruct& target, const string_t& path)
99 {
100 if (!value.is_mapping())
101 {
102 _error(u8"A mapping is expected here.", path);
103 return;
104 }
105
106 const mapping_t& mapping = value.as_mapping();
107
108 std::apply([&](const auto&... fields)
109 {
110 (_load_field(mapping, target, fields, path), ...);
111 }, Binding<Tstruct>::fields);
112
113 _report_unknown_keys<Tstruct>(mapping, path);
114 }

116 template <typename Tstruct, typename Tfield>
117 void _load_field(const mapping_t& mapping, Tstruct& target, const Tfield& field, const string_t& path)
118 {
119 const auto it = mapping.find(field.key);
120
121 // A field the value does not mention keeps what it had.
122 if (it == mapping.end())
123 return;
124
125 _load_value(it->second, target.*(field.member), _extend(path, field.key));
126 }

128 template <typename Tstruct>
129 void _report_unknown_keys(const mapping_t& mapping, const string_t& path)
130 {
131 for (const auto& p : mapping)
132 {
133 bool known = false;
134
135 std::apply([&](const auto&... fields)
136 {
137 ((known = known || fields.key == p.first), ...);
138 }, Binding<Tstruct>::fields);
139
140 if (!known)
141 _warning(u8"Unknown key.", _extend(path, p.first));
142 }

145 template <typename T>
146 void _load_value(const Value& value, T& target, const string_t& path)
147 {
148 if constexpr (bound_struct<T>)
149 _load_struct(value, target, path);
150 else if constexpr (named_enum<T>)
151 _load_enum(value, target, path);
152 else if constexpr (std::same_as<T, string_t>)
153 _load_string(value, target, path);
154 else if constexpr (detail::bound_sequence<T>)
155 _load_sequence(value, target, path);
156 else if constexpr (std::same_as<T, bool>)
157 _load_boolean(value, target, path);
158 else if constexpr (std::integral<T>)
159 _load_integer(value, target, path);
160 else if constexpr (std::floating_point<T>)
161 _load_float(value, target, path);
162 else
163 static_assert(detail::always_false<T>, "This type cannot be bound: give it a `Binding` or an `EnumNames` specialization.");
164 }

166 template <named_enum Tenum>
167 void _load_enum(const Value& value, Tenum& target, const string_t& path)
168 {
169 if (!value.is_string())

176

179 ((found || (names.name == value.as_string() ? (target = names.value, found = true) : false)), ...);
180 }, EnumNames<Tenum>::names);
181
182 if (!found)

213 template <detail::bound_sequence Tsequence>
214 void _load_sequence(const Value& value, Tsequence& target, const string_t& path)
215 {
216 if (!value.is_sequence())
217 {
218 _error(u8"A sequence is expected here.", path);
219 return;
220 }
221
222 // A sequence is replaced rather than added to: a configuration stating a
223 // list means that list, not that list on top of the default one.
224 target.clear();
225
226 const sequence_t& sequence = value.as_sequence();
227
228 for (size_t i = 0; i < sequence.size(); ++i)
229 {
230 typename Tsequence::value_type entry{};
231 _load_value(sequence[i], entry, _index(path, i));
232 target.push_back(std::move(entry));
233 }
234 }

255 template <bound_struct Tstruct>
256 Value _save_struct(const Tstruct& source)
257 {
258 mapping_t mapping;
259
260 std::apply([&](const auto&... fields)
261 {
262 ((mapping[string_t{fields.key}] = _save_value(source.*(fields.member))), ...);

268 template <typename T>
269 Value _save_value(const T& source)
270 {
271 if constexpr (bound_struct<T>)
272 return _save_struct(source);
273 else if constexpr (named_enum<T>)
274 return _save_enum(source);
275 else if constexpr (std::same_as<T, string_t>)
276 return Value{source};
277 else if constexpr (detail::bound_sequence<T>)
278 return _save_sequence(source);
279 else if constexpr (std::same_as<T, bool>)
280 return Value{source};
281 else if constexpr (std::integral<T>)
282 return Value{static_cast<integer_t>(source)};
283 else if constexpr (std::floating_point<T>)
284 return Value{static_cast<float_t>(source)};
285 else
286 static_assert(detail::always_false<T>, "This type cannot be bound: give it a `Binding` or an `EnumNames` specialization.");
287 return {};
288 }

290 template <named_enum Tenum>
291 Value _save_enum(const Tenum& source)
292 {
293 bool found = false;
294 cl7::u8string_view name;
295
296 std::apply([&](const auto&... names)
297 {
298 ((found || (names.value == source ? (name = names.name, found = true) : false)), ...);
299 }, EnumNames<Tenum>::names);
300
301 if (!found)
302 {
303 _error(u8"This enumeration value has no name to write.", {});
304 return {};
305 }
306
307 return Value{name};
308 }

310 template <detail::bound_sequence Tsequence>
311 Value _save_sequence(const Tsequence& source)
312 {
313 sequence_t sequence;
314 sequence.reserve(source.size());
315
316 for (const auto& entry : source)
317 sequence.push_back(_save_value(entry));
318
319 return Value{std::move(sequence)};
