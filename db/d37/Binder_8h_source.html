148 template <bound_struct Tstruct>
149 void _load_struct(const Value& value, Tstruct& target, const string_t& path)
150 {
151 if (!value.is_mapping())
152 {
153 _error(u8"A mapping is expected here.", path);
154 return;
155 }
156
157 const mapping_t& mapping = value.as_mapping();
158
159 std::apply([&](const auto&... fields)
160 {
161 (_load_field(mapping, target, fields, path), ...);
162 }, Binding<Tstruct>::fields);
163
164 _report_unknown_keys<Tstruct>(mapping, path);
165 }

167 template <typename Tstruct, typename Tfield>
168 void _load_field(const mapping_t& mapping, Tstruct& target, const Tfield& field, const string_t& path)
169 {
170 const auto it = mapping.find(field.key);
171
172 // A field the value does not mention keeps what it had.
173 if (it == mapping.end())
174 return;
175
176 _load_value(it->second, target.*(field.member), _extend(path, field.key));
177 }

179 template <typename Tstruct>
180 void _report_unknown_keys(const mapping_t& mapping, const string_t& path)
181 {
182 for (const auto& p : mapping)
183 {
184 bool known = false;
185
186 std::apply([&](const auto&... fields)
187 {
188 ((known = known || fields.key == p.first), ...);
189 }, Binding<Tstruct>::fields);
190
191 if (!known)
192 _warning(u8"Unknown key.", _extend(path, p.first));
193 }

196 template <typename T>
197 void _load_value(const Value& value, T& target, const string_t& path)
198 {
199 if constexpr (detail::bound_optional<T>)
200 _load_optional(value, target, path);
201 else if constexpr (bound_struct<T>)
202 _load_struct(value, target, path);
203 else if constexpr (named_enum<T>)
204 _load_enum(value, target, path);
205 else if constexpr (std::same_as<T, string_t>)
206 _load_string(value, target, path);
207 else if constexpr (detail::bound_sequence<T>)
208 _load_sequence(value, target, path);
209 else if constexpr (detail::bound_map<T>)
210 _load_map(value, target, path);
211 else if constexpr (std::same_as<T, bool>)
212 _load_boolean(value, target, path);
213 else if constexpr (std::integral<T>)
214 _load_integer(value, target, path);
215 else if constexpr (std::floating_point<T>)
216 _load_float(value, target, path);
217 else
218 static_assert(detail::always_false<T>, "This type cannot be bound: give it a `Binding` or an `EnumNames` specialization.");
219 }

227 template <detail::bound_optional Toptional>
228 void _load_optional(const Value& value, Toptional& target, const string_t& path)
229 {
230 if (value.is_null())
231 {
232 target.reset();
233 return;
234 }

236 typename Toptional::value_type inner = target.has_value() ? *target : typename Toptional::value_type{};
237 _load_value(value, inner, path);
238 target = std::move(inner);

288 template <detail::bound_sequence Tsequence>
289 void _load_sequence(const Value& value, Tsequence& target, const string_t& path)
290 {
291 if (!value.is_sequence())
292 {
293 _error(u8"A sequence is expected here.", path);
294 return;
295 }
296
297 // A sequence is replaced rather than added to: a configuration stating a
298 // list means that list, not that list on top of the default one.
299 target.clear();
300
301 const sequence_t& sequence = value.as_sequence();
302
303 for (size_t i = 0; i < sequence.size(); ++i)
304 {
305 typename Tsequence::value_type entry{};
306 _load_value(sequence[i], entry, _index(path, i));
307 target.push_back(std::move(entry));
308 }
309 }

319 template <detail::bound_map Tmap>
320 void _load_map(const Value& value, Tmap& target, const string_t& path)
321 {
322 if (!value.is_mapping())
323 {
324 _error(u8"A mapping is expected here.", path);
325 return;
326 }
327
328 for (const auto& p : value.as_mapping())
329 {
330 auto it = target.find(p.first);
331 if (it == target.end())
332 it = target.emplace(p.first, typename Tmap::mapped_type{}).first;
333
334 _load_value(p.second, it->second, _extend(path, p.first));
335 }

357 template <bound_struct Tstruct>
358 Value _save_struct(const Tstruct& source)
359 {
360 mapping_t mapping;

370 template <typename Tstruct, typename Tfield>
371 void _save_field(mapping_t& mapping, const Tstruct& source, const Tfield& field)
372 {
373 const auto& member = source.*(field.member);
374
375 // Left out entirely rather than written as an explicit null:
376 // a field not set at this level is the normal case, and a
377 // hand-authored file should only ever have to state what differs.
378 if constexpr (detail::bound_optional<std::remove_cvref_t<decltype(member)>>)
379 {
380 if (!member.has_value())
381 return;
382 }
383
384 mapping[string_t{field.key}] = _save_value(member);
385 }

387 template <typename T>
388 Value _save_value(const T& source)
389 {
390 if constexpr (detail::bound_optional<T>)
391 return _save_optional(source);
392 else if constexpr (bound_struct<T>)
393 return _save_struct(source);
394 else if constexpr (named_enum<T>)
395 return _save_enum(source);
396 else if constexpr (std::same_as<T, string_t>)
397 return Value{source};
398 else if constexpr (detail::bound_sequence<T>)
399 return _save_sequence(source);
400 else if constexpr (detail::bound_map<T>)
401 return _save_map(source);
402 else if constexpr (std::same_as<T, bool>)
403 return Value{source};
404 else if constexpr (std::integral<T>)
405 return Value{static_cast<integer_t>(source)};
406 else if constexpr (std::floating_point<T>)
407 return Value{static_cast<float_t>(source)};
408 else
409 static_assert(detail::always_false<T>, "This type cannot be bound: give it a `Binding` or an `EnumNames` specialization.");
410 return {};
411 }
